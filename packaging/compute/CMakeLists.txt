cmake_minimum_required(VERSION 3.16)
project(OpenRevelareCompute LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(WIN32)
  add_library(OpenRevelare.Compute.Native SHARED compute-d3d11.cpp)
  target_link_libraries(OpenRevelare.Compute.Native PRIVATE d3d11 d3dcompiler dxgi)
elseif(APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS version")
  enable_language(OBJCXX)
  add_library(OpenRevelare.Compute.Native SHARED compute-metal.mm)
  target_compile_options(OpenRevelare.Compute.Native PRIVATE -fobjc-arc)
  target_link_libraries(OpenRevelare.Compute.Native PRIVATE "-framework Metal" "-framework Foundation")
  set_target_properties(OpenRevelare.Compute.Native PROPERTIES
    INSTALL_NAME_DIR "@loader_path" BUILD_WITH_INSTALL_NAME_DIR TRUE)
else()
  message(FATAL_ERROR "GPU compute currently supports Windows and macOS")
endif()
target_compile_features(OpenRevelare.Compute.Native PRIVATE cxx_std_17)
install(TARGETS OpenRevelare.Compute.Native RUNTIME DESTINATION . LIBRARY DESTINATION .)
