cmake_minimum_required(VERSION 3.16)
project(OpenRevelareHeif LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Native allocations never cross module ownership; match LibRaw's self-contained CRT.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_PLUGIN_LOADING OFF CACHE BOOL "" FORCE)
set(WITH_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WITH_GDK_PIXBUF OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_DOCUMENTATION OFF CACHE BOOL "" FORCE)
set(WITH_LIBSHARPYUV OFF CACHE BOOL "" FORCE)
foreach(codec LIBDE265 KVAZAAR UVG266 VVDEC VVENC X264 OpenH264_ENCODER OpenH264_DECODER DAV1D AOM_ENCODER AOM_DECODER SvtEnc RAV1E JPEG_ENCODER JPEG_DECODER OpenJPEG_ENCODER OpenJPEG_DECODER FFMPEG_DECODER OPENJPH_ENCODER OPENJPH_DECODER)
  set(WITH_${codec} OFF CACHE BOOL "" FORCE)
endforeach()
set(WITH_X265 ON CACHE BOOL "" FORCE)
add_subdirectory("${HEIF_SOURCE}" heif)
add_library(OpenRevelare.Heif.Native SHARED bridge.cpp)
target_compile_features(OpenRevelare.Heif.Native PRIVATE cxx_std_17)
target_include_directories(OpenRevelare.Heif.Native PRIVATE "${HEIF_SOURCE}/libheif/api" "${CMAKE_CURRENT_BINARY_DIR}/heif")
target_link_libraries(OpenRevelare.Heif.Native PRIVATE heif)
if(UNIX AND NOT APPLE)
  # The codecs are private static dependencies. Bind their symbols locally so
  # x265's direct x86 assembly calls cannot be preempted across the ELF boundary.
  target_link_options(OpenRevelare.Heif.Native PRIVATE "LINKER:--exclude-libs,ALL")
endif()
install(TARGETS OpenRevelare.Heif.Native RUNTIME DESTINATION . LIBRARY DESTINATION .)
